1 · Gọi API và đọc stop_reason

Luôn đọc stop_reason trước khi dùng content. Câu trả lời bị cắt vẫn trả HTTP 200.

import anthropic

client = anthropic.Anthropic()

msg = client.messages.create(
    model="claude-opus-5-5",
    max_tokens=1024,
    system="Bạn là trợ lý hỗ trợ khách hàng.",
    messages=[{"role": "user", "content": ticket}],
)

print(msg.stop_reason)   # end_turn | max_tokens | tool_use | refusal
print(msg.usage)         # input_tokens, output_tokens

if msg.stop_reason == "max_tokens":
    # câu trả lời bị cắt, không phải lỗi: tăng trần rồi chạy lại
    pass
elif msg.stop_reason == "refusal":
    # HTTP 200 nhưng model từ chối
    pass

2 · Streaming

Chữ hiện ngay khi được sinh. SDK ghép sẵn thành message hoàn chỉnh, không tự nối delta.

with client.messages.stream(
    model="claude-opus-5-5",
    max_tokens=2048,
    messages=[{"role": "user", "content": "Giải thích context window."}],
) as stream:
    for event in stream:
        if event.type == "content_block_delta":
            print(event.delta.text, end="", flush=True)
    final = stream.get_final_message()

print(final.stop_reason, final.usage)

3 · Vòng lặp tool

Model không chạy hàm của bạn. Nó trả tool_use, code chạy, gửi tool_result kèm đúng tool_use_id.

tools = [{
    "name": "get_order",
    "description": "Tra cứu trạng thái đơn. Dùng khi khách hỏi về đơn hàng, "
                   "không dùng cho thông tin tài khoản.",
    "input_schema": {
        "type": "object",
        "properties": {"order_id": {"type": "string", "description": "Dạng ANNNNN"}},
        "required": ["order_id"],
        "additionalProperties": False,
    },
    "strict": True,
}]

messages = [{"role": "user", "content": "Đơn A10493 tới đâu rồi?"}]

while True:
    resp = client.messages.create(
        model="claude-opus-5-5", max_tokens=1024,
        messages=messages, tools=tools,
    )
    if resp.stop_reason != "tool_use":
        break
    results = []
    for block in resp.content:
        if block.type == "tool_use":
            try:
                out = run_tool(block.name, block.input)
                results.append({"type": "tool_result",
                                "tool_use_id": block.id, "content": out})
            except Exception as e:
                results.append({"type": "tool_result", "tool_use_id": block.id,
                                "is_error": True, "content": str(e)})
    messages.append({"role": "assistant", "content": resp.content})
    messages.append({"role": "user", "content": results})

4 · Prompt caching

Đặt breakpoint ở cuối phần ổn định. Thứ tự render: tools, system, messages. Cache ngắn hơn mức tối thiểu thì bị bỏ qua im lặng.

resp = client.messages.create(
    model="claude-opus-5-5",
    max_tokens=1024,
    system=[
        {"type": "text", "text": "Bạn là trợ lý hỗ trợ."},
        {"type": "text", "text": policy_document,          # phần lặp lại
         "cache_control": {"type": "ephemeral"}},
    ],
    messages=[{"role": "user", "content": question}],
)

u = resp.usage
print(u.cache_creation_input_tokens, u.cache_read_input_tokens)
# cache_read = 0 nghĩa là không cache được: prefix quá ngắn hoặc có gì đổi

5 · Batch

Rẻ hơn, không stream, kết quả về không theo thứ tự. Ghép bằng custom_id.

requests = [{
    "custom_id": f"ticket-{i}",          # bắt buộc, do bạn đặt
    "params": {
        "model": "claude-opus-5-5",
        "max_tokens": 512,
        "messages": [{"role": "user", "content": t}],
    },
} for i, t in enumerate(tickets)]

job = client.beta.messages.batches.create(requests=requests)

while client.beta.messages.batches.retrieve(job.id).processing_status != "ended":
    time.sleep(30)

for row in client.beta.messages.batches.results(job.id):
    print(row.custom_id, row.result.type)   # rồi ghép theo custom_id

6 · Vision

Ảnh là một block trong content, đặt trước câu hỏi. Ảnh tốn nhiều token hơn chữ; nếu đã có chữ thì gửi chữ.

import base64

with open("receipt.jpg", "rb") as f:
    img = base64.b64encode(f.read()).decode()

resp = client.messages.create(
    model="claude-opus-5-5",
    max_tokens=1024,
    messages=[{
        "role": "user",
        "content": [
            {"type": "image",
             "source": {"type": "base64", "media_type": "image/jpeg", "data": img}},
            {"type": "text", "text": "Trích tổng tiền và ngày, trả JSON."},
        ],
    }],
)

7 · Agent SDK

SDK chạy cả vòng lặp trong process của bạn. Khác API thuần ở chỗ bạn không tự viết loop.

from claude_agent_sdk import ClaudeAgentOptions, query

async for msg in query(
    prompt="Đọc repo này và liệt kê các endpoint chưa có test.",
    options=ClaudeAgentOptions(
        allowed_tools=["Read", "Glob", "Grep"],
        system_prompt="Bạn là kỹ sư review. Chỉ đọc, không sửa file.",
    ),
):
    print(msg)
