def apply_memory_op(store, op):
    kind = op["op"]

    if kind == "NOOP":
        return store

    if kind == "ADD":
        if not any(m["key"] == op["key"] for m in store):
            store.append({"key": op["key"], "value": op["value"]})
        return store

    if kind == "UPDATE":
        for m in store:
            if m["key"] == op["key"]:
                m["value"] = op["value"]
                return store
        store.append({"key": op["key"], "value": op["value"]})
        return store

    if kind == "DELETE":
        store[:] = [m for m in store if m["key"] != op["key"]]
        return store

    raise ValueError(f"Unknown op: {kind}")

import math

def retire_unused(store, use_counts, n):
    threshold = 0.5 * math.log10(n)
    return [m for m in store if use_counts.get(m["key"], 0) >= threshold]
