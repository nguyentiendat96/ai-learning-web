def execution_order(steps):
    deps = {s["id"]: list(s["deps"]) for s in steps}
    for ds in deps.values():
        for dep in ds:
            if dep not in deps:
                raise ValueError(f"dep không tồn tại: {dep}")

    order = []
    done = set()
    remaining = list(deps)
    while remaining:
        ready = [i for i in remaining if all(dep in done for dep in deps[i])]
        if not ready:
            raise ValueError("Có vòng lặp phụ thuộc")
        order.extend(ready)
        done.update(ready)
        remaining = [i for i in remaining if i not in done]
    return order

    done = set()
    remaining = list(deps)
    while remaining:
        ready = [i for i in remaining if all(dep in done for dep in deps[i])]
        if not ready:
            raise ValueError("There is a dependency cycle")
        order.extend(ready)
        done.update(ready)
        remaining = [i for i in remaining if i not in done]
    return order

Danh sách ready ở mỗi vòng chính là các bước "không còn chờ ai" — làm hết chúng rồi mới sang vòng sau. Nếu vòng nào không có ai ready mà vẫn còn bước, chắc chắn có vòng lặp. Đối chiếu với mục 1 của bài giảng: đây là cấu trúc mà agent mua máy thiếu — nó không có chỗ nào ngoài ngữ cảnh để giữ thứ tự và phụ thuộc.The ready list in each round is exactly the set of steps "no longer waiting on anyone" — finish them all, then move to the next round. If a round has nobody ready but steps remain, there is definitely a cycle. Compare with section 1 of the lecture: this is the structure the machine-buying agent lacked — it had nowhere outside its context to keep ordering and dependencies.
