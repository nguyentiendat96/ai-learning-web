def build_mask(messages):
    out = []
    for msg in messages:
        mask = 1 if msg["role"] == "assistant" else 0
        for token in msg["text"].split():
            out.append((token, mask))
    return out

def train_ratio(tokens):
    if not tokens:
        return 0.0
    trained = sum(1 for _, mask in tokens if mask == 1)
    return trained / len(tokens)
