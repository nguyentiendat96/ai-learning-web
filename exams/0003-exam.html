def dispatch(name, args):
    tool = TOOLS.get(name)
    if tool is None:
        raise ValueError(f"Không có tool tên {name}")
    for field in tool["required"]:
        if field not in args:
            raise ValueError(f"Thiếu tham số bắt buộc: {field}")
    for field, value in args.items():
        expected = tool["properties"].get(field)
        if expected is not None and not isinstance(value, expected):
            raise ValueError(f"Tham số {field} sai kiểu")
    return tool["func"](**args)
def dispatch(name, args):
    tool = TOOLS.get(name)
    if tool is None:
        raise ValueError(f"No tool named {name}")
    for field in tool["required"]:
        if field not in args:
            raise ValueError(f"Missing required argument: {field}")
    for field, value in args.items():
        expected = tool["properties"].get(field)
        if expected is not None and not isinstance(value, expected):
            raise ValueError(f"Argument {field} has the wrong type")
    return tool["func"](**args)
